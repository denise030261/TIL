## 코테

### LeetCode 721. Accounts Merge

#### 문제 요약

이름과 메일이 있다. 이름이 같아도 같은 사람일 수도 동명이인일 수 있다. 이 때 만약에 이름도 같고 주어진 배열에서 메일들 중 하나라도 같으면 동일인물이다. 이 때 해당 이름과 그 사람이 가지고 있는 메일들의 배열을 도출하자.

링크 : https://leetcode.com/problems/accounts-merge/description/



#### 첫 번째 시도

처음 문제를 풀 때는 문제를 제대로 해석하지 않아서 같은 이름이면 같은 사람으로 판단하여 메일을 한꺼번에 묶었다.

```
class Solution {
public:
    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {

        vector<vector<string>> answer;
        vector<string> names;
        vector<set<string>> emails;
        
        for(vector<string> account : accounts)
        {
            int n=account.size();
            int m=names.size();
            string name = account[0];
            int exist_idx=-1;

            for(int i=1;i<n;i++)
            {
                string email=account[i];

                for(int j=0;j<m;j++)
                {
                    // 찾음
                    if(emails[j].find(email)!=emails[j].end())
                    {
                        exist_idx=j;
                        break;
                    }
                }

                if(exist_idx!=-1) break;
            }

            for(int i=1;i<n;i++)
            {
                string email=account[i];

                if(exist_idx!=-1)
                {
                    emails[exist_idx].insert(email);
                }
                else if(i==1)
                {
                    set<string> s;
                    s.insert(email);
                    emails.push_back(s);
                    names.push_back(name);
                }
                else
                {
                    emails[emails.size()-1].insert(email);
                }
            }
        }

        for(int i=0;i<names.size();i++)
        {
            vector<string> v;
            v.push_back(names[i]);
            for(auto& email : emails[i])
            {
                v.push_back(email);
            }
            answer.push_back(v);
        }

        return answer;
    }
};
```



#### 두 번째 시도

문제를 제대로 파악하여 이메일을 한 인덱스에서 묶이기 위하여 Union Find 알고리즘을 사용하여 다시 문제를 풀었다. 현재 인덱스의 이메일이 다른 인덱스에 있다는 사실을 알고 연결을 시도했다. 문제는 현재 인덱스의 다른 부분도 다른 인덱스에서 연결을 해야하지만 그렇지 않고 바로 넘겨줬다. 그렇게 하여 Union의 과정이 제대로 이루어지지 않았다.

```
class Solution {
public:
    int root[30001]={0,};
    int Find(int x)
    {
        if(root[x]==x) return x;

        return root[x]=Find(root[x]);
    }

    void Union(int x,int y)
    {
        x=Find(x);
        y=Find(y);

        if(x>y)
        {
            root[x]=y;
        }
        else
        {
            root[y]=x;
        }
    }

    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        unordered_map<string,int> um;
        vector<string> names;
        for(int i=0;i<=30000;i++)
        {
            root[i]=i;
        }

        int index=0;
        for(vector<string> account : accounts)
        {
            bool exist=false;
            int n=account.size();
            int select=0;
            
            for(int i=1;i<n;i++)
            {
                string email=account[i];
                if(um.find(email)!=um.end())
                {
                    // 찾음
                    //cout << account[0] << " : " << email << '\n';
                    Union(index,um[email]);
                    exist=true;
                    select=um[email];
                    break;
                }
                else
                {
                    // 못 찾음
                    um[email]=index;
                }
            }

            if(!exist)
            {
                //cout << "Not Exist "<<account[0] << '\n';
                names.push_back(account[0]);
                index++;
            }
            else
            {
                //cout << "Exist "<<account[0] << '\n';
                for(int i=1;i<n;i++)
                {
                    string email=account[i];
                    um[email]=select;
                }
            }
        }

        vector<vector<string>> answer;
        for(int i=0;i<index;i++)
        {
            vector<string> v;
            vector<string> emails;
            v.push_back(names[i]);

            for(auto& element:um)
            {
                if(element.second==i)
                {
                    emails.push_back(element.first);
                }
            }
            sort(emails.begin(), emails.end());

            for(int j=0;j<emails.size();j++)
            {
                v.push_back(emails[j]);
            }

            answer.push_back(v);
        }

        return answer;
    }
};
```



#### 최종

묶어주는 과정을 끝까지 수행을 하고 역으로 unordered_map에서 인덱스별 이메일을 묶는 과정을 거쳐서 답을 도출하였다. set의 자료구조를 가지는 이유는 이메일의 답안 도출은 정렬을 해야하기 때문이다.

```
class Solution {
public:
    int root[30001]={0,};
    int Find(int x)
    {
        if(root[x]==x) return x;

        return root[x]=Find(root[x]);
    }

    void Union(int x,int y)
    {
        x=Find(x);
        y=Find(y);

        if(x>y)
        {
            root[x]=y;
        }
        else
        {
            root[y]=x;
        }
    }

    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        unordered_map<string,int> um;
        vector<string> names;
        for(int i=0;i<=30000;i++)
        {
            root[i]=i;
        }

        int index=0;
        for(vector<string> account : accounts)
        {
            int n=account.size();
            names.push_back(account[0]);

            for(int i=1;i<n;i++)
            {
                string email=account[i];
                if(um.find(email)!=um.end())
                {
                    // 찾음
                    // cout << account[0] << " : " << email << '\n';
                    Union(index,um[email]);
                }
                else
                {
                    // 못 찾음
                    um[email]=index;
                    //cout << email << ' ' << um[email] << '\n';
                }
            }

            index++;
        }

        vector<vector<string>> answer;
        unordered_map<int,set<string>> um_answer;
        for(auto& element:um)
        {
            int index=Find(element.second);
            if(um_answer.count(index))
            {
                um_answer[index].insert(element.first);
                //있음
            }
            else
            {
                set<string> s;
                s.insert(element.first);
                um_answer[index]=s;
            }
        }

        for(int i=0;i<index;i++)
        {
            vector<string> v;
            if(um_answer.count(i)==0) continue;

            v.push_back(names[i]);
            for(auto& element : um_answer[i])
            {
                v.push_back(element);
            }

            answer.push_back(v);
        }

        return answer;
    }
};
```



#### 나의 생각

만약 문제에서 묶음의 과정이 있으면 그걸 일일이 연결하기 보다는 먼저 Union Find 알고리즘을 먼저 생각을 해야겠다고 생각하였다. 일일이 연결하게 된다면 다른쪽에서 연결하는 과정이 필요하고 그러다보면 어느 한 쪽에서 연결이 안될 때도 있다. 그래서 이를 Union Find 알고리즘을 우선적으로 생각해야겠다.
