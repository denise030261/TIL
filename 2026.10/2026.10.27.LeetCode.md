## 코테

### LeetCode 684. Redundant Connection

#### 문제요약

한 간선을 끊어서 노드끼리 순회하지 않도록 하는 문제

링크 : https://leetcode.com/problems/redundant-connection/description/



#### 첫 번째 시도

순회를 하지 않는다면 한 그룹으로 만들 수 있기 때문에 한 간선씩 빼고 UnionFind 알고리즘을 사용하여 한 그룹이 되는지 확인한다. 만약 한 그룹이 되지 않는다면 그것은 순회가 있다는 것이므로 답은 아닌 것이다. 

```
class Solution {
public:
    int n=0;
    vector<int> root;

    int Find(int x)
    {
        if(x==root[x]) return x;

        return root[x]=Find(root[x]);
    }

    void Union(int x,int y)
    {
        x=Find(x);
        y=Find(y);

        if(root[x]>root[y])
        {
            root[x]=root[y];
        }
        else
        {
            root[y]=root[x];
        }
    }

    void Clear()
    {
        for(int i=0;i<=n;i++)
        {
            root[i]=i;
        }
    }

    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        vector<int> answer;
        n=edges.size();
        int index=0;
        root.push_back(index);
        index++;
        for(vector<int>& edge:edges)
        {
            root.push_back(index);
            index++;
        }

        for(int j=n-1;j>=0;j--)
        {
            vector<int> it=edges[j];

            for(const auto& edge : edges)
            {
                if(edge==it || root[edge[0]]==root[edge[1]]) continue;

                Union(edge[0],edge[1]);
            }

            bool possible=true;
            int standard=root[1];
            for(int i=2;i<=n;i++)
            {
                Union(i,root[i]);

                if(root[i]!=standard)
                {
                    possible=false;
                    break;
                }
            }

            if(possible)
            {
                answer=it;
                break;
            }

            Clear();
        }

        if(answer.size()==0)
        {
            answer=edges[n-1];
        }

        return answer;
    }
};
```



#### 최종

위의 시도를 하니 답은 맞지만 시간이 조금 오래 걸린다. 그래서 다른 로직으로 시도를 하였다. 위의 과정은 간선 하나씩 없애면서 순환 판단한다면 이번에는 간선을 이으면서 만약에 순회가 일어날 때 그 간선이 없애야하는 간선으로 판단하여 이를 출력하도록 하였다.

```
class Solution {
public:
    vector<int> root;

    int Find(int x) 
    {
        if (x == root[x]) return x;

        return root[x] = Find(root[x]); 
    }

    bool Union(int x, int y) 
    {
        int rootX = Find(x);
        int rootY = Find(y);

        if (rootX == rootY) 
        {
            return false; 
        }

        root[rootX] = rootY;
        return true;
    }

    vector<int> findRedundantConnection(vector<vector<int>>& edges) {
        int n = edges.size();
        root.push_back(0);
        for (int i=1;i<=n;i++) 
        {
            root.push_back(i);
        }

        for (const auto& edge : edges) 
        {
            int start=edge[0];
            int end=edge[1];

            if (!Union(start, end)) 
            {
                return edge;
            }
        }

        return {};
    }
};
```



#### 나의 생각

UnionFind알고리즘을 사용하면 무조건 그룹화하고 나서 생각하였는데 그렇게 하지 말고 그룹화하는 과정을 거치면서도 생각을 가져야겠다.
