## 코테

### LeetCode 131. Palindrome Partitioning

#### 문제요약

주어진 문장 중에서 부분 문장으로 나눌 때 모든 부분 문장이 팰린드롬인지 파악하고 맞다면 이를 도출하는 문제

링크 : https://leetcode.com/problems/palindrome-partitioning/description/



#### 첫번째 시도

처음에 문제를 풀을 때 백트레킹까지 생각을 하였지만 부분문자열을 어떻게 해야할지 생각하였다. 부분문자열을 인덱스별로 나누는 방법을 하였지만 중복되는 경우가 있다는 것을 가정하여 set 자료구조를 사용하였다. 근데 그렇게 하니 중복되는 경우에도 재귀를 사용하다보니 시간이 오래 걸렸다.

```
class Solution {
public:
    set<vector<string>> s_answer;
    bool IsPalindrome(string str)
    {
        int n=str.length();
        for(int i=0;i<n/2;i++)
        {
            if(str[i]!=str[n-1-i])
            {
                return false;
            }
        }
        return true;
    }
    
    bool visited[16]={0,};

    void SubString(string s, vector<int>& sub, int index)
    {

        vector<string> v;
        int slice=0;
        for(int i=0;i<sub.size();i++)
        {
            if(sub[i]==0) continue;

            v.push_back(s.substr(slice,sub[i]-slice));
            slice=sub[i];
        }
        v.push_back(s.substr(slice, s.length()-slice));

        bool correct=true;
        for(string str : v)
        {
            if(!IsPalindrome(str))
            {
                correct=false;
                break;
            }
        }
        if(correct) s_answer.insert(v);

        if(sub.size()==s.length())
        {
            return;
        }

        for(int i=index;i<s.length();i++)
        {
            if(!visited[i])
            {
                sub.push_back(i);
                visited[i]=true;
                SubString(s,sub,i);
                visited[i]=false;
                sub.pop_back();
            }
        }
    }

    vector<vector<string>> partition(string s) {
        vector<vector<string>> answer;
        vector<int> v;
        SubString(s,v,0);

        for(auto& element : s_answer)
        {
            answer.push_back(element);
        }
        return answer;
    }
};
```



#### 두번째시도

두번째 시도에는 증복의 과정을 삭제하였다. 하지만 그래도 시간이 오래 걸렸다. 그 이유를 확인해보니 팰린드롬이 아닌 문자열이 있으면 그 이후도 팰린드롬이 아닌데 필요없는 과정을 처리하였기 때문에 시간이 걸렸다.

```
class Solution {
public:
    vector<vector<string>> answer;
    bool IsPalindrome(string& str)
    {
        int n=str.length();
        for(int i=0;i<n/2;i++)
        {
            if(str[i]!=str[n-1-i])
            {
                return false;
            }
        }
        return true;
    }

    void SubString(string s, vector<int>& sub, int index)
    {

        vector<string> v;
        int slice=0;
        for(int i=0;i<sub.size();i++)
        {
            if(sub[i]==0) continue;

            v.push_back(s.substr(slice,sub[i]-slice));
            slice=sub[i];
        }
        v.push_back(s.substr(slice, s.length()-slice));

        bool correct=true;
        for(string str : v)
        {
            if(!IsPalindrome(str))
            {
                correct=false;
                break;
            }
        }
        if(correct) answer.push_back(v);

        if(sub.size()==s.length())
        {
            return;
        }

        for(int i=index;i<s.length();i++)
        {
            sub.push_back(i);
            SubString(s,sub,i+1);
            sub.pop_back();
        }
    }

    vector<vector<string>> partition(string s) {
        vector<int> v;
        SubString(s,v,1);
        return answer;
    }
};
```



#### 최종

팰린드롬이 아닌 경우에는 생략을 하였지만 만약에 마지막 문자열이 팰린드롬이 아닌 경우에는 쪼갤 수 있는 경우가 있기 때문에 그대로 진행하는 방식으로 구현하였다.

```
class Solution {
public:
    vector<vector<string>> answer;
    bool IsPalindrome(string& str)
    {
        int n=str.length();
        for(int i=0;i<n/2;i++)
        {
            if(str[i]!=str[n-1-i])
            {
                return false;
            }
        }
        return true;
    }

    void SubString(string& s, vector<string>& v, int slice)
    {
        if (slice == s.length())
        {
            answer.push_back(v);
            return;
        }

        for (int i = slice + 1; i <= s.length(); i++)
        {
            string str = s.substr(slice, i - slice);

            if (!IsPalindrome(str))
                continue;

            v.push_back(str);
            SubString(s, v, i);
            v.pop_back();
        }
    }

    vector<vector<string>> partition(string s) {
        vector<string> v;
        SubString(s,v,0);
        return answer;
    }
};
```
