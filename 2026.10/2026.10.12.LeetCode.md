## 코테

### LeetCode 934. Shortest Bridge

#### 문제 요약

1은 땅이고 0은 바다일 때, 섬(땅과 잇는 것)과 섬과의 거리를 구해라

링크 : https://leetcode.com/problems/shortest-bridge/description/



#### 첫 번째 시도

섬을 구할 때 bfs로 구하고 섬과 섬의 거리를 구할 때는 priority_queue을 이용하여 bfs로 구하였다. 그 결과 시간이 오래 걸렸다. 16ms 걸렸다.

```
class Solution {
public:
    int dx[4]={0,0,-1,1};
    int dy[4]={1,-1,0,0};
    int n=-1;
    vector<pair<int,int>> islands;

    void Island(vector<vector<int>>& grid, int area, pair<int,int> cur)
    {
        queue<pair<int,int>> q;
        q.push(cur);
        while(!q.empty())
        {
            int x=q.front().first;
            int y=q.front().second;
            q.pop();

            for(int i=0;i<4;i++)
            {
                int next_x=x+dx[i];
                int next_y=y+dy[i];
                if(next_x>=0 && next_x<n && next_y>=0 && next_y<n)
                {
                    if(grid[next_x][next_y]==1)
                    {
                        grid[next_x][next_y]=area;
                        q.push({next_x,next_y});
                        if(area==2) islands.push_back({next_x,next_y});
                    }
                }
            }
        }
    }

    int shortestBridge(vector<vector<int>>& grid) {
        n=grid.size();

        // 섬 나누기
        int area=2;
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<n;j++)
            {
                if(grid[i][j]==1)
                {
                    grid[i][j]=area;
                    if(area==2) islands.push_back({i,j});
                    Island(grid,area,{i,j});
                    area++;
                }
            }
        }

        priority_queue<pair<int,pair<int,int>>> pq;
        vector<vector<bool>> visited(n,vector<bool>(n,false));
        for(int i=0;i<islands.size();i++)
        {
            int x=islands[i].first;
            int y=islands[i].second;
            visited[x][y]=true;
            pq.push({-0,{x,y}});
        }

        int answer=0;
        while(!pq.empty())
        {
            int cost=-pq.top().first;
            int x=pq.top().second.first;
            int y=pq.top().second.second;
            pq.pop();

            if(grid[x][y]==3) 
            {
                answer=cost;
                break;
            }
            
            for(int i=0;i<4;i++)
            {
                int next_x=x+dx[i];
                int next_y=y+dy[i];
                if(next_x>=0 && next_x<n && next_y>=0 && next_y<n)
                {
                    if(grid[next_x][next_y]!=2 && !visited[next_x][next_y])
                    {
                        visited[next_x][next_y]=true;
                        pq.push({-(cost+1),{next_x,next_y}});
                    }
                }
            }
        }
        return answer-1;
    }
};
```



#### 최종

문제는 섬을 하나만 구해서 다른 땅까지 구하면 되는데 섬 2개를 bfs로 구했다는 점, 큐를 bfs에서 구하는 과정 중에 담아도 되는데 따로 구했다는 점과 queue로 구해도 먼저 도달하면 그 거리가 짧은 거리임에도 불구하고 priority_queue로 구해서 시간이 오래 걸렸다 다시 구했을 때는 0ms걸렸다.

```
class Solution {
public:
    int dx[4]={0,0,-1,1};
    int dy[4]={1,-1,0,0};
    int n=-1;
    queue<pair<int,pair<int,int>>> pq;

    void Island(vector<vector<int>>& grid, int area, pair<int,int> cur)
    {
        queue<pair<int,int>> q;
        q.push(cur);
        while(!q.empty())
        {
            int x=q.front().first;
            int y=q.front().second;
            q.pop();

            for(int i=0;i<4;i++)
            {
                int next_x=x+dx[i];
                int next_y=y+dy[i];
                if(next_x>=0 && next_x<n && next_y>=0 && next_y<n)
                {
                    if(grid[next_x][next_y]==1)
                    {
                        grid[next_x][next_y]=area;
                        q.push({next_x,next_y});
                        pq.push({-0,{next_x,next_y}});
                    }
                }
            }
        }
    }

    int shortestBridge(vector<vector<int>>& grid) {
        n=grid.size();

        // 섬 나누기
        bool divide=false;
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<n;j++)
            {
                if(grid[i][j]==1)
                {
                    grid[i][j]=2;
                    pq.push({-0,{i,j}});
                    Island(grid,2,{i,j});
                    divide=true;
                    break;
                }
            }

            if(divide) break;
        }

        int answer=INT_MAX;
        vector<vector<bool>> visited(n,vector<bool>(n,false));
        while(!pq.empty())
        {
            int cost=-pq.front().first;
            int x=pq.front().second.first;
            int y=pq.front().second.second;
            pq.pop();

            if(grid[x][y]==1) 
            {
                answer=cost-1;
                break;
            }
            
            for(int i=0;i<4;i++)
            {
                int next_x=x+dx[i];
                int next_y=y+dy[i];
                if(next_x>=0 && next_x<n && next_y>=0 && next_y<n)
                {
                    if(grid[next_x][next_y]!=2 && !visited[next_x][next_y])
                    {
                        visited[next_x][next_y]=true;
                        pq.push({-(cost+1),{next_x,next_y}});
                    }
                }
            }
        }
        return answer;
    }
};
```
