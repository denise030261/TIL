### LeetCode 722. Remove Comments

#### 문제 요약

주석인 '//'와 '/* */'의 부분을 지우고 문자열을 반환하는 문제

링크 : https://leetcode.com/problems/remove-comments/description/



#### 첫 번째 시도

처음 문제를 풀 때는 /*, */ 을 처리했을 때 for문을 주석 다음 부분으로 넘어가야 했는데 주석의 두 번째 상태로 넘겨주었다.

```
class Solution {
public:
    vector<string> removeComments(vector<string>& source) {
        vector<string> answer;

        bool comment=false;
        string answer_str="";
        for(string str : source)
        {
            int n=str.length();
            for(int i=0;i<n;i++)
            {
                if(str[i]=='/' && i+1<n && str[i+1]=='/')
                {
                    break;        
                }
                else if(str[i]=='/' && i+1<n && str[i+1]=='*')
                {
                    comment=true;
                    continue;
                }
                else if(comment && str[i]=='*' && i+1<n && str[i+1]=='/')
                {
                    comment=false;
                    i++;
                    continue;
                }
                else if(comment)
                {
                    continue;
                }

                answer_str+=str[i];
            }

            if(answer_str!="" && !comment)
            {
                answer.push_back(answer_str);
                answer_str="";
            }
        }

        return answer;
    }
};
```

#### 최종

위의 두 부분을 보완해서 아래와 같은 코드를 도출하였다.

```
class Solution {
public:
    int root[30001]={0,};
    int Find(int x)
    {
        if(root[x]==x) return x;

        return root[x]=Find(root[x]);
    }

    void Union(int x,int y)
    {
        x=Find(x);
        y=Find(y);

        if(x>y)
        {
            root[x]=y;
        }
        else
        {
            root[y]=x;
        }
    }

    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        unordered_map<string,int> um;
        vector<string> names;
        for(int i=0;i<=30000;i++)
        {
            root[i]=i;
        }

        int index=0;
        for(vector<string> account : accounts)
        {
            int n=account.size();
            names.push_back(account[0]);

            for(int i=1;i<n;i++)
            {
                string email=account[i];
                if(um.find(email)!=um.end())
                {
                    // 찾음
                    // cout << account[0] << " : " << email << '\n';
                    Union(index,um[email]);
                }
                else
                {
                    // 못 찾음
                    um[email]=index;
                    //cout << email << ' ' << um[email] << '\n';
                }
            }

            index++;
        }

        vector<vector<string>> answer;
        unordered_map<int,set<string>> um_answer;
        for(auto& element:um)
        {
            int index=Find(element.second);
            if(um_answer.count(index))
            {
                um_answer[index].insert(element.first);
                //있음
            }
            else
            {
                set<string> s;
                s.insert(element.first);
                um_answer[index]=s;
            }
        }

        for(int i=0;i<index;i++)
        {
            vector<string> v;
            if(um_answer.count(i)==0) continue;

            v.push_back(names[i]);
            for(auto& element : um_answer[i])
            {
                v.push_back(element);
            }

            answer.push_back(v);
        }

        return answer;
    }
};
```
