### LeetCode 211. Design Add and Search Words Data Structure

#### 문제 요약

이 문제는 단어를 추가하고 찾는 문제이다. 이 때 단어를 찾을 때 '.'는 어떤 문자가 들어가든 상관없다

링크 : https://leetcode.com/problems/design-add-and-search-words-data-structure/description/


#### 첫 번째 시도

처음에는 깊게 생각하지 않고 한 번 '.' 부분에 소문자만 들어가니 소문자를 대입해서 map 자료구조를 이용하여 찾는 로직으로 구하였다. 그렇게 하니 시간이 초과된다.

```
class WordDictionary {
public:
    set<string> s;
    WordDictionary() {
        s.clear();
    }
    
    void addWord(string word) {
        s.insert(word);
    }
    
    bool search(string word) {
        vector<int> dots;
        int n=word.size();
        for(int i=0;i<n;i++)
        {
            if(word[i]=='.')
            {
                dots.push_back(i);
            }
        }

        int m=dots.size();
        if(m==0)
        {
            if(s.find(word)!=s.end()) return true;

            return false;
        }
        else if(m==1)
        {
            int index=dots[0];
            for(char c='a';c<='z';c++)
            {
                word[index]=c;
                if(s.find(word)!=s.end()) return true;
            }

            return false;
        }
        else
        {
            int index1=dots[0];
            int index2=dots[1];

            for(char c1='a';c1<='z';c1++)
            {
                for(char c2='a';c2<='z';c2++)
                {
                    word[index1]=c1;
                    word[index2]=c2;

                    if(s.find(word)!=s.end()) return true;
                }
            }
        }

        return false;
    }
};

/**
 * Your WordDictionary object will be instantiated and called as such:
 * WordDictionary* obj = new WordDictionary();
 * obj->addWord(word);
 * bool param_2 = obj->search(word);
 */
```



#### 두 번째 시도

시간을 단축하기 위해서 어떻게 풀어야 고민을 했을 때 트라이를 사용하였다. 트라이를 사용하여 만약 '.' 보게 된다면 for문을 통해서 모든 문자를 돌아서 확인하는 식으로 로직을 구하였다. 근데 계속 틀리다고 떴다.

class WordDictionary {
public:
struct Trie{
    map<char,Trie*> m;
    string word;
};
    set<string> s;
    Trie* root = nullptr;

    void dfs(Trie* cur, int index, string str)
    {
        if(index==str.length())
        {
            cur->word=str;
            return;
        }
        char c=str[index];
        
        if(cur->m.find(c)==cur->m.end())
        {
            Trie* next=new Trie();
            cur->m[c]=next;
        }

        dfs(cur->m[c],index+1,str);
    }

    bool find_dfs(Trie* cur,int index, string str)
    {
        if(cur==nullptr) return false;
        char c=str[index];
        if(index==str.length())
        {
            if(ㄴㅅ==cur->word)
            {
                return true;
            }
            return false;
        }

        if(c=='.')
        {
            if(cur->m.empty())
            {
                return false;
            }

            for(auto& element:cur->m)
            {
                if(find_dfs(element.second,index+1,str))
                {
                    return true;
                }
            }
        }
        else if(cur->m.find(c)!=cur->m.end())
        {
            if(find_dfs(cur->m[c],index+1,str))
            {
                return true;
            }
        }
        return false;
    }

    WordDictionary() {
        s.clear();
        root=new Trie();
    }
    
    void addWord(string word) {
        s.insert(word);
        dfs(root,0,word);
    }
    
    bool search(string word) {
        return find_dfs(root,0,word);
    }
};

#### 최종

다시 문제를 보니 단어가 '.'가 포함되니 문자열로 비교하는 것이 아닌 이 노드에서 끝나는지를 확인해야 했다. 또한, 소문자만 있으니 굳이 map이나 unordered_map 없이 배열을 사용하여 문제를 풀어도 되었다. 

```
  class WordDictionary {
public:
struct Trie{
    Trie* next[26] = {nullptr};
    bool isEnd=false;
};
    Trie* root = nullptr;

    void dfs(Trie* cur, int index, string& str)
    {
        if(index==str.length())
        {
            cur->isEnd=true;
            return;
        }
        char c=str[index];
        
        if(cur->next[c-'a']==nullptr)
        {
            Trie* next=new Trie();
            cur->next[c-'a']=next;
        }

        dfs(cur->next[c-'a'],index+1,str);
    }

    bool find_dfs(Trie* cur,int index, string& str)
    {
        if(cur==nullptr) return false;
        if (index == str.length())
        {
            return cur->isEnd;
        }

        char c = str[index];

        if(c=='.')
        {
            for(int i=0;i<26;i++)
            {
                if(find_dfs(cur->next[i],index+1,str))
                {
                    return true;
                }
            }
        }
        else if(cur->next[c-'a']!=nullptr)
        {
            if(find_dfs(cur->next[c-'a'],index+1,str))
            {
                return true;
            }
        }
        return false;
    }

    WordDictionary() {
        root=new Trie();
    }
    
    void addWord(string word) {
        dfs(root,0,word);
    }
    
    bool search(string word) {
        return find_dfs(root,0,word);
    }
};

/**
 * Your WordDictionary object will be instantiated and called as such:
 * WordDictionary* obj = new WordDictionary();
 * obj->addWord(word);
 * bool param_2 = obj->search(word);
 */
```
