## LeetCode 339. Evaluate Division

### 문제요약

식이 주어질 때 그 식의 나눗셈의 값이 주어진다. 그 식들을 이용해서 다른 식의 값을 구해라.

링크 : https://leetcode.com/problems/evaluate-division/



### 첫 번째 시도

처음에는 아예 감이 안 오다가 힌트를 보았다. 힌트를 봐서 그래프 방식으로 풀어야하겠다는 것을 알았다. 문제는 bc, cd 이런 식이었는데 이 식은 b*c가 아닌 bc 자체가 변수이기 때문에 따로 곱하거나 할 필요는 없다.

```
class Solution {
public:
    vector<double> answer;
    unordered_map<string,unordered_map<string,double>> um;

    double dfs(string exp1,string exp2)
    {
        if(um[exp1].find(exp2)!=um[exp1].end()) return um[exp1][exp2];

        for(auto& element1 : um[exp1])
        {
            if(um[element1.first].find(exp2)!=um[element1.first].end()) return um[element1.first][exp2];

            for(auto& element2 : um[exp2])
            {
                if(um[element1.first].find(element2.first)!=um[element1.first].end()) return um[element1.first][element2.first];
            }
        }

        return -1.0;
    }

    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {
        int n=equations.size();
        for(int i=0;i<n;i++)
        {
            vector<string> equation=equations[i];
            string q=equation[0];
            string a=equation[1];
            double value=values[i];
            unordered_map<string,int> ratio;
            ratio[a]=value;

            um[q][a]=value;
            um[a][q]=1.0f/value;
            um[q][q]=1.0f;
            um[a][a]=1.0f;
        }

        for(auto element : um)
        {
            string str=element.first;
            unordered_map<string,double> um_ele=element.second;
            cout << str << '\n';
            for(auto element2 : um_ele)
            {
                cout << element2.first << ' ' << element2.second << '\n';
            }
            cout << '\n';
        }

        for(vector<string> query : queries)
        {
            string exp1=query[0];
            string exp2=query[1];

            if(um.find(exp1)==um.end())
            {
                answer.push_back(-1.0);
            }
            else
            {
                answer.push_back(dfs(exp1,exp2));
            }
        }


        return answer;
    }
};
```



### 두 번째 시도

그 다음은 단순한 실수와 재귀에서 틀리게 풀었다. 일단 처음에는 double이어야 하는데 int로 자료구조를 잘못 구했고 dfs에서 더 깊이 들어가야 풀 수 있는 것을 하나의 깊이로만 풀어서 풀지를 못하였다

또한, 이전 코드에서는 분자 따로, 분모 따로 하여 하드코딩을 구하려고 했으나 그렇게 하면 식이 복잡하고 이미 비율로 저장을 하였으니 하나의 비율로 재귀를 구하는 방식이 좋다.

```
class Solution {
public:
    vector<double> answer;
    unordered_map<string,unordered_map<string,double>> um;
    unordered_map<string,bool> visited;

    double dfs(string current,string target, double product)
    {
        if(current==target) return product;

        visited[current]=true;
        for(auto& element : um[current])
        {
            string next_node=element.first;
            double weight=element.second;

            if(!visited[next_node])
            {
                double result=dfs(next_node, target, product*weight);
                if(result!=-1.0)
                {
                    return result;
                }
            }
        }

        return -1.0;
    }

    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {
        int n=equations.size();
        for(int i=0;i<n;i++)
        {
            vector<string> equation=equations[i];
            string q=equation[0];
            string a=equation[1];
            double value=values[i];

            um[q][a]=value;
            um[a][q]=1.0/value;
            um[q][q]=1.0;
            um[a][a]=1.0;
        }

        for(vector<string> query : queries)
        {
            string exp1=query[0];
            string exp2=query[1];

            if(um.find(exp1)==um.end() || um.find(exp2)==um.end())
            {
                answer.push_back(-1.0);
            }
            else if(exp1==exp2)
            {
                answer.push_back(1.0);
            }
            else
            {
                visited.clear();
                answer.push_back(dfs(exp1,exp2,1.0));
            }
        }


        return answer;
    }
};
```



### 나의 생각

코드 문제를 풀 때는 어렵게 생각하지 않고 간단하고 최대한 하드코딩하지 않은 방식으로 풀어야겠다고 생각이 들었다. 오히려 하드코딩으로 하려니 생각이 꼬이게 된다.
