## 코테

### LeetCode 698. Partition to K Equal Sum Subsets

#### 문제 요약

주어진 배열을 k만큼 나누어서 그 합이 같은지 판단하는 문제

링크 : https://leetcode.com/problems/partition-to-k-equal-sum-subsets/description/



#### 첫 번째 시도

처음 이 문제를 풀 때는 dfs와 비트마스킹을 사용해 문제를 풀었다. 문제는 풀 수 있지만 시간 초과의 문제를 걸렸다. 비트 마스킹을 사용하게 된다면 2^N에 dfs까지 3^N의 시간까지 걸려 시간 초과 걸린다.

```
class Solution {
public:
    bool dfs(vector<int>& nums,int k,int visited, vector<int>& v)
    {
        if(v.size()==k)
        {
            if(visited!=(1<<nums.size())-1)
            {
                return false;
            }

            int sum=0;
            for(int i=0;i<k;i++)
            {
                int temp=0;
                for(int j=0;j<nums.size();j++)
                {
                    if((v[i])&(1<<j))
                    {
                        temp+=nums[j];
                    }
                }
                //cout << v[i] << ' ';
                //cout << sum << ' ' << temp << '\n';
                if(i==0)
                {
                    sum=temp;
                }
                else if(sum!=temp)
                {
                    return false;
                }
            }
            //cout << "true\n ";
            return true;
        }

        for(int i=0;i<(1<<nums.size());i++)
        {
            if((visited & i)==0)
            {
                v.push_back(i);
                if(dfs(nums,k,visited|i,v))
                {
                    return true;
                }
                v.pop_back();
            }
        }

        return false;
    }
    bool canPartitionKSubsets(vector<int>& nums, int k) {
        vector<int> v;
        return dfs(nums,k,0,v);
    }
};
```



#### 두번째 시도

비트마스킹이 아닌 백트래킹 방식을 진행하였다. 그래서 k를 미리 만들어서 배분하는 방식인데 0~nums.size()까지 돌아서 다시 시간 초과가 났다. 

```
class Solution {
public:
    bool dfs(vector<int>& nums, int k,int target,vector<bool>& visited,vector<int>& v)
    {
        bool answer=true;
        for(int i=0;i<k;i++)
        {   
            if(v[i]!=target)
            {
                answer=false;
                break;
            }
        }
        if(answer) return true;

        for(int i=0;i<nums.size();i++)
        {
            if(!visited[i])
            {
                int num=nums[i];

                for(int j=0;j<k;j++)
                {
                    if(num+v[j]<=target)
                    {
                        v[j]+=num;
                        visited[i]=true;
                        if(dfs(nums,k,target,visited,v))
                        {
                            return true;
                        }
                        visited[i]=false;
                        v[j]-=num;
                    }
                }
            }
        }

        return false;
    }

    bool canPartitionKSubsets(vector<int>& nums, int k) {
        sort(nums.rbegin(),nums.rend());
        int target=0;
        for(int num:nums)
        {
            target+=num;
        }

        if(target%k!=0)
        {
            return false;
        }
        target/=k;

        vector<int> v(k,0);
        vector<bool> visited(nums.size(),false);
        return dfs(nums,k,target,visited,v);
    }
};
```



#### 최종

#### 나의 생각

입력 범위가 작고 집합 문제로 생각하여 바로 비트마스킹을 생각하였지만 그 시간까지 고려하지 못하였다. 빈 곳을 어떻게 넣을 수 있을까를 너무 비트마스킹으로만 제한하여서 생각해서 더 생각하지 못하였다. 좀 더 러프한 생각으로 문제를 풀어야겠다.
