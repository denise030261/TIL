## 코테

### LeetCode 310. Minimum Height Trees

#### 문제 요약

높이가 가장 낮은 트리를 만들기 위해서 어떤 root를 가져야하는지에 대한 출력을 구하는 문제다

링크 : https://leetcode.com/problems/minimum-height-trees/description/



#### 첫번째 시도

처음 이 문제를 볼 때는 dfs나 bfs로 풀면 되겠다고 생각을 하였다. 하지만 입력값이 커서 시간 초과가 걸렸다.

```
class Solution {
public:
    vector<int> findMinHeightTrees(int n, vector<vector<int>>& edges) {
        vector<int> answer;
        vector<int> v[n];

        if(edges.size()==1) return edges[0];

        for(vector<int> edge : edges)
        {
            int start=edge[0];
            int end=edge[1];

            v[start].push_back(end);
            v[end].push_back(start);
        }

        vector<bool> visited(n,false);
        vector<bool> miss(n,0);
        int dist=INT_MAX;
        for(int i=0;i<n;i++)
        {
            if(miss[i]) continue;
            for(int j=0;j<n;j++)
            {
                visited[j]=false;
            }

            vector<int> area(n,0);
            visited[i]=true;
            int depth=0;
            queue<pair<int,int>> q;
            q.push({i,0});
            area[i]=0;
            while(!q.empty())
            {
                int cur=q.front().first;
                int cost=q.front().second;
                depth=cost;
                area[cur]=cost;
                q.pop();

                for(int j=0;j<v[cur].size();j++)
                {
                    int next=v[cur][j];
                    if(!visited[next])
                    {
                        visited[next]=true;
                        q.push({next,cost+1});
                    }
                }
            }
            
            for(int j=0;j<n;j++)
            {
                if(area[j]==depth)
                {
                    miss[j]=true;
                }
            }

            if(dist==depth)
            {
                answer.push_back(i);
            }
            else if(dist>depth)
            {
                dist=depth;
                answer.clear();
                answer.push_back(i);
            }
        }

        return answer;
    }
};
```



#### 두번째 시도

그 다음에는 힌트를 보았는데 높이가 낮은 값을 가지는 루트를 가지기 위해서는 트리를 쭉 늘어났을 때 중간에 있는 노드가 루트가 되는 값이다. 이 값을 구하기 위해서 위상 정렬을 사용하였다. 위상 정렬을 사용하면 리프부터 중간노드까지 들어갈 수 있기 때문에 이로 구하였다. 

```
class Solution {
public:
    vector<int> findMinHeightTrees(int n, vector<vector<int>>& edges) {
        vector<int> v[n];
        vector<int> connects(n,0);
        vector<int> answer;

        if(edges.size()==0)
        {
            for(int i=0;i<n;i++)
            {
                answer.push_back(i);
            }

            return answer;
        }

        for(vector<int> edge : edges)
        {
            int start=edge[0];
            int end=edge[1];
        
            connects[start]++;
            v[start].push_back(end);
            connects[end]++;
            v[end].push_back(start);
        }

        queue<pair<int,int>> q;
        for(int i=0;i<n;i++)
        {
            if(connects[i]==1)
            {
                connects[i]--;
                q.push({i,0});
            }
        }

        int depth=0;
        while(!q.empty())
        {
            int cur=q.front().first;
            int cost=q.front().second;
            q.pop();
            if(cost>depth)
            {
                depth=cost;
                answer.clear();
                answer.push_back(cur);
            }
            else if(depth==cost)
            {
                answer.push_back(cur);
            }

            for(int i=0;i<v[cur].size();i++)
            {
                int next=v[cur][i];
                connects[next]--;

                if(connects[next]==1)
                {
                    q.push({next,cost+1});
                }
            }
        }
        return answer; 
    }
};
```



#### 최종

풀었긴 하였지만 평균적인 시간보다 오래 걸렸다. 그 이유를 살펴보니 

    for(vector<int> edge : edges)

부분에서 벡터를 복사했기 때문에 시간이 걸렸다.

```
class Solution {
public:
    vector<int> findMinHeightTrees(int n, vector<vector<int>>& edges) {
        if (n == 1) return {0};
        if (n == 2) return {0, 1};

        vector<int> v[n];
        vector<int> connects(n,0);
        vector<int> answer;

        for(const auto& edge : edges)
        {
            int start=edge[0];
            int end=edge[1];
        
            connects[start]++;
            v[start].push_back(end);
            connects[end]++;
            v[end].push_back(start);
        }

        queue<int> q;
        int remain=n;
        for(int i=0;i<n;i++)
        {
            if(connects[i]==1)
            {
                q.push(i);
            }
        }

        int depth=0;
        while(remain>2)
        {
            int qz=q.size();
            remain-=qz;

            for(int i=0;i<qz;i++)
            {
                int cur=q.front();
                q.pop();

                for(int j=0;j<v[cur].size();j++)
                {
                    int next=v[cur][j];
                    connects[next]--;
                    if(connects[next]==1)
                    {
                        q.push(next);
                    }
                }
            }
        }

        while(!q.empty())
        {
            answer.push_back(q.front());
            q.pop();
        }

        return answer; 
    }
};
```



#### 나의 생각

매개변수의 참조를 생각하였고 for문에서는 이를 생각하지 않았다. 이 부분에 대해서도 생각을 해야했다. 또한, 위상정렬은 무조건 단방향으로만 해야한다는 오해는 하면 안되겠다고 생각하였다.
