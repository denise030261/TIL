- 상수 버퍼 선언 교체
  
  

```
VSOutput VSMain(VSInput input)
{
    VSOutput output;

    // 1. 크기 변경
    float2 scaledPosition = input.position.xy * objectScale;

    // 2. 원점 기준 회전
    float c = cos(rotation);
    float s = sin(rotation);

    float2 rotatedPosition;

    rotatedPosition.x =
        scaledPosition.x * c - scaledPosition.y * s;

    rotatedPosition.y =
        scaledPosition.x * s + scaledPosition.y * c;

    // 3. 화면 비율 보정
    rotatedPosition.x /= aspectRatio;

    // 4. 객체 위치로 이동
    float2 finalPosition = rotatedPosition + moveOffset.xy;

    output.position = float4(
        finalPosition,
        input.position.z + moveOffset.z,
        1.0f
    );

    output.color = input.color;

    return output;
}
```

#### 4. Update() 전체 교체

```
void Update(float deltaTime)
{
    if (g_selectedHouse >= g_houses.size())
    {
        return;
    }

    House& house = g_houses[g_selectedHouse];

    // 이동 방향
    float directionX = 0.0f;
    float directionY = 0.0f;

    if (g_keyDown[VK_LEFT])  directionX -= 1.0f;
    if (g_keyDown[VK_RIGHT]) directionX += 1.0f;
    if (g_keyDown[VK_UP])    directionY += 1.0f;
    if (g_keyDown[VK_DOWN])  directionY -= 1.0f;

    const float length = std::sqrt(
        directionX * directionX +
        directionY * directionY
    );

    if (length > 0.0f)
    {
        directionX /= length;
        directionY /= length;

        // 화면에서 가로·세로 이동 속도도 맞추기
        house.transform.offsetX +=
            directionX * house.moveSpeed * deltaTime / g_aspectRatio;

        house.transform.offsetY +=
            directionY * house.moveSpeed * deltaTime;
    }

    // 회전: 초당 약 90도
    const float rotationSpeed = 1.5707963f;

    if (g_keyDown['Q'])
    {
        house.transform.rotation += rotationSpeed * deltaTime;
    }

    if (g_keyDown['E'])
    {
        house.transform.rotation -= rotationSpeed * deltaTime;
    }

    // 각도가 계속 큰 값으로 누적되지 않도록 정리
    const float twoPi = 6.2831853f;

    house.transform.rotation =
        std::fmod(house.transform.rotation, twoPi);

    // 크기 변경
    const float scaleSpeed = 0.3f;

    if (g_keyDown['Z'])
    {
        house.transform.scale -= scaleSpeed * deltaTime;
    }

    if (g_keyDown['X'])
    {
        house.transform.scale += scaleSpeed * deltaTime;
    }

    // 너무 작아지거나 커지지 않도록 제한
    if (house.transform.scale < 0.05f)
    {
        house.transform.scale = 0.05f;
    }

    if (house.transform.scale > 1.5f)
    {
        house.transform.scale = 1.5f;
    }
}
```

| 각도   | 라디안        |
| ---- | ---------- |
| 90도  | 약 `1.5708` |
| 180도 | 약 `3.1416` |
| 360도 | 약 `6.2832` |
