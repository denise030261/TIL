## CS(DirectX)

### 여러 객체로 움직이기

#### 1. 집 객체와 저장 공간 만들기

```
#include <vector>

struct House
{
    TransformData transform{};
    float moveSpeed = 0.3f;
};

std::vector<House> g_houses;
std::size_t g_selectedHouse = 0;
```



#### 2. 집 두 채 생성하기

```
void InitScene();

void InitScene()
{
    g_houses.clear();

    House leftHouse;
    leftHouse.transform.offsetX = -0.5f;

    House rightHouse;
    rightHouse.transform.offsetX = 0.5f;

    g_houses.push_back(leftHouse);
    g_houses.push_back(rightHouse);

    g_selectedHouse = 0;
}

InitScene();

ShowWindow(hWnd, nCmdShow);
UpdateWindow(hWnd);
```



#### 3. 두 채가 들어가도록 집 크기 줄이기

```
float3 scaledPosition = input.position * 0.35f;
float3 movedPosition = scaledPosition + moveOffset.xyz;
```



#### 4. 숫자 키로 조작할 집 선택하기

```
case WM_KEYDOWN:
{
    if (wParam < g_keyDown.size())
    {
        g_keyDown[static_cast<std::size_t>(wParam)] = true;
    }

    // 키보드 상단 숫자 키
    if (wParam == '1' && !g_houses.empty())
    {
        g_selectedHouse = 0;
    }
    else if (wParam == '2' && g_houses.size() >= 2)
    {
        g_selectedHouse = 1;
    }

    return 0;
}
```



#### 5. 선택한 집의 위치만 갱신하기

```
if (g_selectedHouse >= g_houses.size())
{
    return;
}

House& selectedHouse = g_houses[g_selectedHouse]; // House로만 하면 복사본이기 때문에 복사 비용이 발생한다

selectedHouse.transform.offsetX +=
    directionX * selectedHouse.moveSpeed * deltaTime;

selectedHouse.transform.offsetY +=
    directionY * selectedHouse.moveSpeed * deltaTime;
```



#### 6. 집마다 이동량을 전달하고 그리기

```
// 모든 집이 같은 상수 버퍼를 사용
ID3D11Buffer* constantBuffer = g_constantBuffer.Get();

g_context->VSSetConstantBuffers(
    0,
    1,
    &constantBuffer
);

// 각 집의 위치를 전달하고 그리기
for (const House& house : g_houses)
{
    g_context->UpdateSubresource(
        g_constantBuffer.Get(),
        0,
        nullptr,
        &house.transform,
        0,
        0
    );

    g_context->DrawIndexed(g_indexCount, 0, 0);
}

// 모든 집을 그린 뒤 한 번 표시
return g_swapChain->Present(1, 0);
```
