## 코테

### LeetCode 983. Minimum Cost For Tickets

#### 문제 요약

1일, 7일, 30일 티켓이 있다. 기차 여행할 day들이 있을 때 가장 싼 가격으로 티켓을 끊어야 하는 문제

링크 : https://leetcode.com/problems/minimum-cost-for-tickets/description/



#### 첫 번째 요약

dp와 2차 for문을 사용하여 문제를 풀었다. 현재를 기준으로 안에 있는 for문을 통해서 그 이전의 day와 비교하여 싼 가격의 티켓을 dp 방식으로 구한다.

```
class Solution {
public:
    int mincostTickets(vector<int>& days, vector<int>& costs) {
        int n=days.size();
        vector<int> dp(n+1,INT_MAX);
        dp[0]=0;

        for(int i=0;i<n;i++)
        {
            if(days[i]<=7)
            {
                dp[i+1]=min(dp[i+1],costs[1]);
            }
            if(days[i]<=30)
            {
                dp[i+1]=min(dp[i+1],costs[2]);
            }

            for(int j=0;j<i;j++)
            {
                if(days[i]-days[j]<7)
                {
                    dp[i+1]=min(dp[i+1],dp[j]+costs[1]);
                }
                if(days[i]-days[j]<30)
                {
                    dp[i+1]=min(dp[i+1],dp[j]+costs[2]);
                }
            }

            dp[i+1]=min(dp[i+1],dp[i]+min(costs[0],min(costs[1],costs[2])));
        }

        return dp[n];
    }
};
```



#### 다른 풀이법1

dp를 풀 때 투포인터도 사용하여 문제를 푸는 방식이다. 7일티켓용 포인터, 30일티켓용 포인터를 잡아 현재에서 각 7일, 30일 간격 이내를 유지하면서 dp방식으로 문제를 푼다. 

```
class Solution {
public:
    int mincostTickets(vector<int>& days, vector<int>& costs) {
        int n = days.size();
        vector<int> dp(n + 1, 0);
        int idx7 = 0, idx30 = 0;

        for (int i = 0; i < n; ++i) {
            // days[i]까지 커버 가능한 가장 이른 시작 인덱스 찾기
            while (days[i] - days[idx7] >= 7) ++idx7;
            while (days[i] - days[idx30] >= 30) ++idx30;

            int cost1 = dp[i] + costs[0];
            int cost7 = dp[idx7] + costs[1];
            int cost30 = dp[idx30] + costs[2];

            dp[i + 1] = min({cost1, cost7, cost30});
        }

        return dp[n];
    }
};
```

#### 다른 풀이법2

입력값이 많이 작기 때문에 배열을 인덱스 대신에 실제 날짜로 상태를 잡아서 dp 방식으로 답을 구한다

```
class Solution {
public:
    int mincostTickets(vector<int>& days, vector<int>& costs) {
        int lastDay = days.back();
        vector<bool> isTravelDay(lastDay + 1, false);
        for (int d : days) {
            isTravelDay[d] = true;
        }

        // dp[d]: 1일부터 d일까지 여행하는 데 드는 최소 비용
        vector<int> dp(lastDay + 1, 0);

        for (int d = 1; d <= lastDay; ++d) {
            // 여행하지 않는 날은 전날 최소 비용 그대로 유지
            if (!isTravelDay[d]) {
                dp[d] = dp[d - 1];
                continue;
            }

            // 여행하는 날: 1일권, 7일권, 30일권 중 최소 비용 선택
            int cost1 = dp[d - 1] + costs[0];
            int cost7 = dp[max(0, d - 7)] + costs[1];
            int cost30 = dp[max(0, d - 30)] + costs[2];

            dp[d] = min({cost1, cost7, cost30});
        }

        return dp[lastDay];
    }
};
```



#### 나의 생각

주어진 입력이 작기 때문에 2중문을 사용하여 문제를 풀었다. 입력이 클 것을 대비해서 문제 푸는 습관도 가져야겠다.
