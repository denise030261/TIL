## 코테

### LeetCode 210. Course Schedule 2

#### 문제 요약

수강 과목을 전체를 듣기 위해 어떤 순서로 들어야하는지 출력하는 문제

링크 : https://leetcode.com/problems/course-schedule-ii/description/



#### 첫 번째 시도

이 문제는 먼저 완료를 해야하는 작업이 있어서 생각했던 것은 위상 정렬이었다. 그래서 edges가 0이 되면 다음 수강을 듣도록 함과 동시에 답 배열에 넣는다. 이를 반복하면서 만약 그 배열의 크기가 수강의 크기와 같으면 그대로 출력하지만 아니면 빈 배열을 출력한다

```
class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int> answer;
        vector<int> edges(numCourses,0);
        vector<int> v[numCourses];

        for(vector<int>& prerequisite : prerequisites)
        {
            int first_sub=prerequisite[1];
            int second_sub=prerequisite[0];

            edges[second_sub]++;
            v[first_sub].push_back(second_sub);
        }

        queue<int> q;
        for(int i=0;i<numCourses;i++)
        {
            if(edges[i]==0)
            {
                answer.push_back(i);
                q.push(i);
            }
        }

        while(!q.empty())
        {
            int cur=q.front();
            q.pop();

            for(int i=0;i<v[cur].size();i++)
            {
                int next=v[cur][i];
                edges[next]--;

                if(edges[next]==0)
                {
                    answer.push_back(next);
                    q.push(next);
                }
            }
        }

        if(answer.size()!=numCourses)
        {
            return {};
        }
        return answer;
    }
};
```



#### 최종

조금 더 시간을 줄이기 위해서 answer 배열과 큐를 나누지 않고 answer로 for문을 이용해서 위상정렬 알고리즘을 사용하였다.

```
class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int> answer;
        vector<int> edges(numCourses,0);
        vector<int> v[numCourses];

        for(int i=0;i<prerequisites.size();i++)
        {
            int first_sub=prerequisites[i][1];
            int second_sub=prerequisites[i][0];

            edges[second_sub]++;
            v[first_sub].push_back(second_sub);
        }

        for(int i=0;i<numCourses;i++)
        {
            if(edges[i]==0)
            {
                answer.push_back(i);
            }
        }

        for(int i=0;i<answer.size();i++)
        {
            int cur=answer[i];

            for(int j=0;j<v[cur].size();j++)
            {
                int next=v[cur][j];
                edges[next]--;

                if(edges[next]==0)
                {
                    answer.push_back(next);
                }
            }
        }

        return (answer.size()!=numCourses?vector<int>{}:answer);
    }
};
```



#### 나의 생각

이번 문제를 계기로 위상 정렬의 흔한 템플릿이 아닌 다른 방식으로 접근하였다. 잘하는 알고리즘은 다른 양상으로 문제를 풀 수 있는지 고민을 가져보는 것도 좋을 것 같다.
