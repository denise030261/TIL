## 코테

### LeetCode 909. Snakes and Ladders

#### 문제 요약

보드가 있고 사다리로 앞으로 건너뛰고 뱀을 탄다면 뒤로 간다. 마지막 지점에 도달할 때 최소한의 움직임은 몇인가

링크 : https://leetcode.com/problems/snakes-and-ladders/description/



#### 첫 번째 시도

처음에는 dp를 사용하여 풀었다. 하지만 dp의 경우에는 작은 문제를 풀어 큰 문제로 도달하는 문제인데 뱀의 경우에는 역방향으로 점프를 하기 때문에 dp 알고리즘에 대해 위배가 된다. 

```
class Solution {
public:
    int snakesAndLadders(vector<vector<int>>& board) {
        // init
        int n=board.size();
        vector<int> dp(n*n+1,INT_MAX);
        dp[0]=0;
        dp[1]=0;
        int walk=1;

        if(board[0][0]!=-1) return -1;

        while(walk<=n*n)
        {
            int move=walk;
            // row의 홀수는 ->, 짝수는 <-
            for(int i=1;i<=6;i++)
            {
                move++;
                
                int row=(n-1)-move/n; 
                int col=row%2==0?(n-1)-(move%n-1):move%n-1;

                if(move%n==0)
                {
                    row++;
                    col=col==n?col-1:col+1;
                }

                if(move>n*n) break;
                dp[move]=min(dp[move],dp[walk]+1);
                
                int num=board[row][col];
                if(num!=-1)
                {
                    dp[num]=min(dp[num],dp[move]);
                }
            }

            walk++;
        }

        // for(int i=1;i<=n*n;i++)
        // {
        //     cout <<i<<' '<< dp[i] << '\n';
        // }

        return dp[n*n];
    }
};
```



#### 최종

다음으로 생각한 방법이 bfs이다. 움직이는 비용이 1이기 때문에 우선순위 큐가 아닌 큐를 사용하여 문제를 풀었고 먼저 마지막 지점을 도달한 것이 적은 비용을 도출하는 것이다. 

```
class Solution {
public:
    int snakesAndLadders(vector<vector<int>>& board) {
        int n=board.size();
        int answer=-1;

        queue<pair<int,int>> q;
        vector<int> nums;
        vector<bool> visited;
        bool forward=true;

        nums.push_back(0);
        visited.push_back(false);
        for(int i=n-1;i>=0;i--)
        {
            int col=0;
            if(forward)
            {
                for(int j=0;j<n;j++)
                {
                    nums.push_back(board[i][j]);
                    visited.push_back(false);
                }
            }
            else
            {
                for(int j=n-1;j>=0;j--)
                {
                    nums.push_back(board[i][j]);
                    visited.push_back(false);
                }
            }
            forward=!forward;
        }

        q.push({1,0});
        visited[1]=true;
        while(!q.empty())
        {
            int cur=q.front().first;
            int cost=q.front().second;
            q.pop();

            if(cur==n*n)
            {
                answer=cost;
                break;
            }

            for(int i=1;i<=6 && cur+i<=n*n;i++)
            {
                int next=cur+i;
                int destination=(nums[next]==-1)?next:nums[next];

                if(!visited[destination]) 
                {
                    visited[destination] = true;
                    q.push({destination,cost+1});
                }
            }
        }

        return answer;
    }
};
```



#### 다른 사람의 문제 풀이

다른 사람도 bfs로 풀었지만 본인의 문제 풀이와 다른 점이 있다면 필요한 위치만 좌표 계산을 하였다는 점이다. 그래서 본인의 문제 풀이보다 더 빠르게 걸린다.

```
class Solution {
public:
    // 1차원 번호(num)를 2차원 좌표(row, col)로 변환하는 함수
    pair<int, int> getCoord(int num, int n) {
        int r = (num - 1) / n;
        int c = (num - 1) % n;
        int row = (n - 1) - r; // 아래에서 위로
        // 짝수 행인지 홀수 행인지에 따라 열 방향 결정
        int col = (row % 2 == (n - 1) % 2) ? c : (n - 1 - c);
        return {row, col};
    }

    int snakesAndLadders(vector<vector<int>>& board) {
        int n = board.size();
        vector<bool> visited(n * n + 1, false);
        queue<pair<int, int>> q; // {현재 칸 번호, 이동 횟수}

        q.push({1, 0});
        visited[1] = true;

        while (!q.empty()) {
            auto [curr, cost] = q.front();
            q.pop();

            if (curr == n * n) {
                return cost;
            }

            for (int i = 1; i <= 6 && curr + i <= n * n; i++) {
                int next = curr + i;
                auto [r, c] = getCoord(next, n);
                
                int destination = (board[r][c] == -1) ? next : board[r][c];

                if (!visited[destination]) {
                    visited[destination] = true;
                    visited[next] = true; // 주사위로 밟은 칸도 방문 처리
                    q.push({destination, cost + 1});
                }
            }
        }

        return -1;
    }
};
```
